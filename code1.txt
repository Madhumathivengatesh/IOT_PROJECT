# UNKNOWN FACE HASH FUNCTIONS
# ============================================================

def make_hash(face_image):

    """
    Create a simple perceptual hash
    for duplicate unknown-face detection.
    """

    if face_image.size == 0:

        return None


    # --------------------------------------------------------
    # Convert to grayscale
    # --------------------------------------------------------

    if len(face_image.shape) == 3:

        gray = cv2.cvtColor(
            face_image,
            cv2.COLOR_BGR2GRAY
        )

    else:

        gray = face_image


    # --------------------------------------------------------
    # Resize
    # --------------------------------------------------------

    small = cv2.resize(
        gray,
        (16, 16)
    )


    # --------------------------------------------------------
    # Calculate average
    # --------------------------------------------------------

    average = small.mean()


    # --------------------------------------------------------
    # Convert to bits
    # --------------------------------------------------------

    bits = (
        small >= average
    ).flatten()


    return bits


def hash_distance(
    hash1,
    hash2
):

    if hash1 is None or hash2 is None:

        return 999


    return int(
        (hash1 != hash2).sum()
    )


# ============================================================
# LOAD EXISTING UNKNOWN HASHES
# ============================================================

unknown_hashes = []


for filename in os.listdir(
    UNKNOWN_FOLDER
):

    if not filename.lower().endswith(
        ".jpg"
    ):

        continue


    image_path = os.path.join(
        UNKNOWN_FOLDER,
        filename
    )


    image = cv2.imread(
        image_path
    )


    if image is None:

        continue


    image_hash = make_hash(
        image
    )


    if image_hash is not None:

        unknown_hashes.append(
            image_hash
        )


# ============================================================
# SAVE UNKNOWN FACE
# ============================================================

def save_unknown_face(
    face_image
):

    global unknown_hashes


    if face_image is None:

        return


    if face_image.size == 0:

        return


    new_hash = make_hash(
        face_image
    )


    if new_hash is None:

        return


    # --------------------------------------------------------
    # Check duplicate
    # --------------------------------------------------------

    for old_hash in unknown_hashes:

        distance = hash_distance(
            new_hash,
            old_hash
        )


        if distance <= UNKNOWN_HASH_DISTANCE:

            # Same/similar unknown face
            return


    # --------------------------------------------------------
    # Create real date and time
    # --------------------------------------------------------

    now = datetime.now()


    timestamp = now.strftime(
        "%Y%m%d_%H%M%S_%f"
    )


    filename = (
        f"unknown_{timestamp}.jpg"
    )


    file_path = os.path.join(
        UNKNOWN_FOLDER,
        filename
    )


    # --------------------------------------------------------
    # Save image
    # --------------------------------------------------------

    success = cv2.imwrite(
        file_path,
        face_image
    )


    if success:

        unknown_hashes.append(
            new_hash
        )


        print()
        print(
            "UNKNOWN PERSON DETECTED"
        )


        print(
            "Image:",
            file_path
        )


        print(
            "Date:",
            now.strftime("%Y-%m-%d")
        )


        print(
            "Time:",
            now.strftime("%H:%M:%S")
        )

employee_state = {}


# ============================================================
# PROCESS IN/OUT CROSSING
# ============================================================

def process_employee_crossing(
    employee_id,
    center_y
):

    current_timestamp = time.time()


    # --------------------------------------------------------
    # Determine current side of line
    # --------------------------------------------------------

    if center_y < LINE_Y:

        current_side = "above"

    else:

        current_side = "below"


    # --------------------------------------------------------
    # First time seeing employee
    # --------------------------------------------------------

    if employee_id not in employee_state:

        employee_state[
            employee_id
        ] = {

            "side": current_side,

            "status": "OUT",

            "last_event": 0.0
        }


        return "OUT"


    previous_side = employee_state[
        employee_id
    ]["side"]


    # --------------------------------------------------------
    # Update current side
    # --------------------------------------------------------

    employee_state[
        employee_id
    ]["side"] = current_side


    # --------------------------------------------------------
    # Cooldown
    # --------------------------------------------------------

    last_event = employee_state[
        employee_id
    ]["last_event"]


    if (
        current_timestamp - last_event
        < CROSSING_COOLDOWN
    ):

        return employee_state[
            employee_id
        ]["status"]


    # --------------------------------------------------------
    # Detect downward crossing
    # --------------------------------------------------------

    crossed_down = (

        previous_side == "above"

        and current_side == "below"
    )


    # --------------------------------------------------------
    # Detect upward crossing
    # --------------------------------------------------------

    crossed_up = (

        previous_side == "below"

        and current_side == "above"
    )


    event = None


    # ========================================================
    # DETERMINE IN / OUT DIRECTION
    # ========================================================

    if IN_DIRECTION == "down":

        if crossed_down:

            event = "IN"

        elif crossed_up:

            event = "OUT"


    else:

        if crossed_up:

            event = "IN"

        elif crossed_down:

            event = "OUT"


    # --------------------------------------------------------
    # No crossing
    # --------------------------------------------------------

    if event is None:

        return employee_state[
            employee_id
        ]["status"]


    # --------------------------------------------------------
    # Update employee state
    # --------------------------------------------------------

    employee_state[
        employee_id
    ]["status"] = event


    employee_state[
        employee_id
    ]["last_event"] = current_timestamp


    # --------------------------------------------------------
    # Real date/time
    # --------------------------------------------------------

    now = datetime.now()


    date_text = now.strftime(
        "%Y-%m-%d"
    )


    time_text = now.strftime(
        "%H:%M:%S"
    )


    # --------------------------------------------------------
    # Terminal output
    # --------------------------------------------------------

    print()
    print(
        "----------------------------------------------"
    )


    print(
        f"Employee ID : {employee_id}"
    )


    print(
        f"Date        : {date_text}"
    )


    print(
        f"Time        : {time_text}"
    )


    print(
        f"Operation   : {event}"
    )


    print(
        "----------------------------------------------"
    )


    return event

    # --------------------------------------------------------

        gray,

            MIN_FACE_HEIGHT

    # Draw crossing line
    # --------------------------------------------------------

        frame,
        (0, LINE_Y),
        (
            frame.shape[1],
            LINE_Y
        ),
        (255, 255, 0),
        2
    )


    cv2.putText(
        frame,
        "ENTRY / EXIT LINE",
        (20, LINE_Y - 10),
        cv2.FONT_HERSHEY_SIMPLEX,
        0.6,
        (255, 255, 0),
        2
    )


    # ========================================================
    # PROCESS DETECTED FACES
    # ========================================================

    for (x, y, w, h) in faces:


        # ----------------------------------------------------
        # Check minimum size
        # ----------------------------------------------------

        if w < MIN_FACE_WIDTH:

            continue


        if h < MIN_FACE_HEIGHT:

            continue


        # ----------------------------------------------------
        # Face centre
        # ----------------------------------------------------

        center_x = (
            x + w // 2
        )

        center_y = (
            y + h // 2
        )


        # ----------------------------------------------------
        # Crop grayscale face
        # ----------------------------------------------------

        face = gray[
            y:y+h,
            x:x+w
        ]


        if face.size == 0:

            continue


        # ----------------------------------------------------
        # Resize
        # ----------------------------------------------------

        face_resized = cv2.resize(
            face,
            (200, 200)
        )


        # ----------------------------------------------------
        # LBPH prediction
        # ----------------------------------------------------

        employee_id, confidence = (
            recognizer.predict(
                face_resized
            )
        )


        # ====================================================
        # KNOWN EMPLOYEE
        # ====================================================

        if (

            confidence < RECOGNITION_THRESHOLD

            and employee_id in registered_ids
        ):


            status = process_employee_crossing(
                employee_id,
                center_y
            )


            label = (
                f"ID {employee_id} | "
                f"{status}"
            )


            box_color = (
                0,
                255,
                0
            )


        # ====================================================
        # UNKNOWN PERSON
        # ====================================================

        else:


            label = "UNKNOWN"


            box_color = (
                0,
                0,
                255
            )


            # ------------------------------------------------
            # Save colour face
            # ------------------------------------------------

            color_face = frame[
                y:y+h,
                x:x+w
            ]


            save_unknown_face(
                color_face
            )


        # ----------------------------------------------------
        # Draw face box
        # ----------------------------------------------------

        cv2.rectangle(
            frame,
            (x, y),
            (x+w, y+h),
            box_color,
            2
        )


        # ----------------------------------------------------
        # Draw center point
        # ----------------------------------------------------

        cv2.circle(
            frame,
            (
                center_x,
                center_y
            ),
            4,
            box_color,
            -1
        )


        # ----------------------------------------------------
        # Display label
        # ----------------------------------------------------

        cv2.putText(
            frame,
            label,
            (
                x,
                max(y - 10, 20)
            ),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.7,
            box_color,
            2
        )


    # --------------------------------------------------------
    # Display camera
    # --------------------------------------------------------

    cv2.imshow(
        "24/7 Office Face Monitoring",
        frame
    )


    # --------------------------------------------------------
    # Quit with Q
    # --------------------------------------------------------

    if cv2.waitKey(1) & 0xFF == ord("q"):

        break
